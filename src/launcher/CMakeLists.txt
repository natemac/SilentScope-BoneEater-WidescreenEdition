cmake_minimum_required(VERSION 3.20)
project(BoneEaterPrototypeLauncher LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The native prototype launcher requires Windows and MSVC.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(launcher_core STATIC launcher_model.cpp launcher_win32.cpp launcher_close.cpp calibration.cpp player_data.cpp display.cpp)
target_include_directories(launcher_core PUBLIC .)
set(BONE_EATER_UPSTREAM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/spice2x/src/spice2x" CACHE PATH "Pinned upstream headers")
target_include_directories(launcher_core SYSTEM PRIVATE "${BONE_EATER_UPSTREAM_ROOT}")
target_compile_features(launcher_core PUBLIC cxx_std_20)
target_compile_options(launcher_core PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(launcher_core PUBLIC bcrypt)
add_executable(bone_eater_launcher WIN32 main.cpp)
target_link_libraries(bone_eater_launcher PRIVATE launcher_core shell32)
target_compile_options(bone_eater_launcher PRIVATE /W4 /permissive- /EHsc /utf-8)
set_target_properties(bone_eater_launcher PROPERTIES OUTPUT_NAME "Bone Eater WS-Edition")
enable_testing()
add_executable(launcher_fixture ../../tests/launcher/launcher_fixture.cpp)
target_compile_features(launcher_fixture PRIVATE cxx_std_20)
target_compile_options(launcher_fixture PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(launcher_tests ../../tests/launcher/launcher_tests.cpp)
target_link_libraries(launcher_tests PRIVATE launcher_core shell32)
target_include_directories(launcher_tests PRIVATE ..)
target_compile_options(launcher_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
add_test(NAME launcher_tests COMMAND launcher_tests $<TARGET_FILE:launcher_fixture>)
add_executable(launcher_close_fixture ../../tests/launcher/launcher_close_fixture.cpp)
target_include_directories(launcher_close_fixture BEFORE PRIVATE ../../tests/render/reticle_stubs ../../src)
target_compile_features(launcher_close_fixture PRIVATE cxx_std_20)
target_compile_options(launcher_close_fixture PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(launcher_close_fixture PRIVATE bcrypt comctl32)
add_executable(launcher_close_tests ../../tests/launcher/launcher_close_tests.cpp)
target_link_libraries(launcher_close_tests PRIVATE launcher_core)
target_compile_options(launcher_close_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
add_test(NAME launcher_close_tests COMMAND launcher_close_tests $<TARGET_FILE:launcher_close_fixture> $<TARGET_FILE:bone_eater_launcher>)

add_executable(calibration_tests ../../tests/launcher/calibration_tests.cpp)
target_link_libraries(calibration_tests PRIVATE launcher_core)
add_test(NAME calibration_tests COMMAND calibration_tests)

add_executable(display_tests ../../tests/launcher/display_tests.cpp)
target_link_libraries(display_tests PRIVATE launcher_core)
add_test(NAME display_tests COMMAND display_tests)
add_executable(display_probe ../../tests/launcher/display_probe.cpp)
target_link_libraries(display_probe PRIVATE launcher_core)
add_executable(player_data_tests ../../tests/launcher/player_data_tests.cpp)
target_link_libraries(player_data_tests PRIVATE launcher_core)
add_test(NAME player_data_tests COMMAND player_data_tests)
