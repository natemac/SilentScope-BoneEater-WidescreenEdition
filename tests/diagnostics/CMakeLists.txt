cmake_minimum_required(VERSION 3.16)
project(BoneEaterDiagnosticPolicyTests LANGUAGES CXX)
enable_testing()
if(NOT MSVC)
    message(FATAL_ERROR "These fixtures exercise the Windows production callback/SEH modules")
endif()
add_executable(policy_tests policy_tests.cpp)
target_include_directories(policy_tests PRIVATE ../../src)
target_compile_features(policy_tests PRIVATE cxx_std_17)
target_compile_options(policy_tests PRIVATE /W4 /permissive- /EHsc)
foreach(value absent 1 0 true 01 12345)
    add_test(NAME policy_${value} COMMAND policy_tests ${value})
endforeach()
foreach(kind CAMERA FRAMING DOF)
    string(TOLOWER "${kind}" name)
    add_executable(${name}_output_tests functional_output_tests.cpp ../../src/camera/vertical_pan_policy.cpp)
    target_include_directories(${name}_output_tests BEFORE PRIVATE stubs ../../src)
    target_compile_definitions(${name}_output_tests PRIVATE TEST_${kind}=1)
    target_compile_features(${name}_output_tests PRIVATE cxx_std_20)
    target_compile_options(${name}_output_tests PRIVATE /W4 /permissive- /EHsc /utf-8)
    foreach(mode normal quiet directory open seek tell header)
        add_test(NAME ${name}_output_${mode} COMMAND ${name}_output_tests ${mode})
    endforeach()
endforeach()
