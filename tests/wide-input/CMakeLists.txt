cmake_minimum_required(VERSION 3.16)
project(BoneEaterWideInputTests LANGUAGES CXX)
enable_testing()
add_executable(native_wide_input_tests native_wide_input_tests.cpp)
target_include_directories(native_wide_input_tests PRIVATE ../../src)
target_compile_features(native_wide_input_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(native_wide_input_tests PRIVATE /W4 /permissive- /EHsc /UNDEBUG)
else()
    target_compile_options(native_wide_input_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME native_wide_input_tests COMMAND native_wide_input_tests)
