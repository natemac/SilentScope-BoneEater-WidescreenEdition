cmake_minimum_required(VERSION 3.16)
project(BoneEaterSelectedHidTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(selected_hid_tests selected_hid_tests.cpp)
target_include_directories(selected_hid_tests PRIVATE ../../src)
target_compile_features(selected_hid_tests PRIVATE cxx_std_17)
target_link_libraries(selected_hid_tests PRIVATE Threads::Threads)
if(WIN32)
    target_compile_definitions(selected_hid_tests PRIVATE BONE_EATER_SELECTED_HID_TESTING)
    target_sources(selected_hid_tests PRIVATE ../../src/input/selected_hid.cpp)
    target_link_libraries(selected_hid_tests PRIVATE hid user32)
    # Explicit manually run diagnostic; never registered as an automatic test.
    # The production reader is compiled without its testing macro here.
    add_executable(selected_hid_probe ../../tools/selected_hid_probe.cpp ../../src/input/selected_hid.cpp)
    target_include_directories(selected_hid_probe PRIVATE ../../src)
    target_compile_features(selected_hid_probe PRIVATE cxx_std_17)
    target_compile_definitions(selected_hid_probe PRIVATE NOMINMAX)
    target_link_libraries(selected_hid_probe PRIVATE hid user32 Threads::Threads)
    if(MSVC)
        target_compile_options(selected_hid_probe PRIVATE /W4 /permissive- /EHsc)
    endif()
endif()
if(MSVC)
    target_compile_options(selected_hid_tests PRIVATE /W4 /permissive- /EHsc /UNDEBUG)
else()
    target_compile_options(selected_hid_tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME selected_hid_tests COMMAND selected_hid_tests)
