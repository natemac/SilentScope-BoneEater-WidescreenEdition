cmake_minimum_required(VERSION 3.16)
project(BoneEaterStandaloneTests LANGUAGES CXX)
enable_testing()
add_executable(input_profile_tests input_profile_tests.cpp
    ../../src/input/selected_hid_config.cpp ../../src/input/source_policy.cpp)
target_include_directories(input_profile_tests PRIVATE ../../src)
target_include_directories(input_profile_tests SYSTEM PRIVATE ../../vendor/spice2x/src/spice2x)
target_compile_features(input_profile_tests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(input_profile_tests PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(input_profile_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME input_profile_tests COMMAND input_profile_tests)
