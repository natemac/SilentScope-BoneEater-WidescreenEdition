cmake_minimum_required(VERSION 3.16)
project(BoneEaterCheckedTargetTests LANGUAGES CXX)
if(NOT MSVC)
    message(FATAL_ERROR "These isolated Windows memory-permission tests require MSVC SEH.")
endif()
enable_testing()
add_executable(checked_target_tests checked_target_tests.cpp)
target_include_directories(checked_target_tests PRIVATE ../../src)
target_compile_features(checked_target_tests PRIVATE cxx_std_17)
target_compile_options(checked_target_tests PRIVATE /W4 /permissive-)
add_test(NAME checked_target_tests COMMAND checked_target_tests)
